// Cálculo ajustado para la caja real const calcularEfectivoCaja = () => { const totalIngresosCobrados = movimientos .filter(m => m.tipo === 'ingreso') .reduce((acc, m) => acc + m.monto, 0); const totalGastosVarios = movimientos .filter(m => m.tipo === 'gasto') .reduce((acc, m) => acc + m.monto, 0); const totalInvertidoRepuestos = trabajosPorCobrar .reduce((acc, t) => acc + t.costoInvertido, 0); return totalIngresosCobrados - totalGastosVarios - totalInvertidoRepuestos; };
